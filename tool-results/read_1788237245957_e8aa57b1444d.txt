    75→    75→export function BookingDialog({

    80→    80→  presetRoomTypeId,

    83→    83→}: BookingDialogProps) {

    85→    85→
    86→    86→  const [step, setStep] = useState(1)
    87→    87→  const [search, setSearch] = useState<SearchParams>(initialSearch)
    88→    88→  const [editOpen, setEditOpen] = useState(false)
    89→    89→  const [items, setItems] = useState<AvailabilityItem[]>([])
    90→    90→  const [searching, setSearching] = useState(false)
    91→    91→  const [searchError, setSearchError] = useState<string | null>(null)
    92→    92→  const [selected, setSelected] = useState<AvailabilityItem | null>(null)
    93→    93→  const [guest, setGuest] = useState<GuestForm>(EMPTY_GUEST)
    94→    94→  const [policyAccepted, setPolicyAccepted] = useState(false)
    95→    95→  const [errors, setErrors] = useState<Record<string, string>>({})
    96→    96→  const [paymentMethod, setPaymentMethod] = useState<'PAY_AT_HOTEL' | 'CARD'>('PAY_AT_HOTEL')
    97→    97→  const [confirmed, setConfirmed] = useState<ReservationPublic | null>(null)
    98→    98→
    99→    99→  // آخر قيم props (تُقرأ عند الفتح دون إعادة تشغيل التأثيرات)
   100→   100→  const initialSearchRef = useRef(initialSearch)
   101→   101→  initialSearchRef.current = initialSearch
   102→   102→  const presetRef = useRef(presetRoomTypeId)
   103→   103→  presetRef.current = presetRoomTypeId
   104→   104→
   105→   105→  // مفتاح idempotency — يولد مرة لكل تدفق حجز ويثبت
   106→   106→  const idemKeyRef = useRef('')
   107→   107→  const submittingRef = useRef(false)
   108→   108→
   109→   109→  // ── البحث ──
   110→   110→  const runSearch = useCallback(
   111→   111→    async (s: SearchParams): Promise<AvailabilityItem[]> => {
   112→   112→      setSearching(true)
   113→   113→      setSearchError(null)
   114→   114→      try {
   115→   115→        const res = await api<{ items: AvailabilityItem[] }>('/api/public/availability', {
   116→   116→          method: 'POST',
   117→   117→          body: s,
   118→   118→        })
   119→   119→        setItems(res.items)
   120→   120→        return res.items
   121→   121→      } catch (e) {

   123→   123→        const msg = e instanceof ApiError ? e.message : 'تعذر البحث عن الغرف — يرجى المحاولة مرة أخرى'
   124→   124→        setSearchError(msg)
   125→   125→        return []
   126→   126→      } finally {
   127→   127→        setSearching(false)
   128→   128→      }
   129→   129→    },
   130→   130→    []

   133→   133→  // ── إعادة التعيين + البحث التلقائي عند الفتح ──

   139→   139→    setSearch(s)
   140→   140→    setEditOpen(false)
   141→   141→    setItems([])
   142→   142→    setSearchError(null)
   143→   143→    setSelected(null)
   144→   144→    setGuest(EMPTY_GUEST)
   145→   145→    setPolicyAccepted(false)
   146→   146→    setErrors({})
   147→   147→    setPaymentMethod('PAY_AT_HOTEL')
   148→   148→    setConfirmed(null)
   149→   149→    idemKeyRef.current = ''
   150→   150→    submittingRef.current = false
   151→   151→
   152→   152→    let cancelled = false
   153→   153→    void runSearch(s).then((result) => {
   154→   154→      if (cancelled) return
   155→   155→      // تحديد مسبق لنوع الغرفة (زر «احجز الآن» من بطاقة الغرفة)
   156→   156→      if (preset) {
   157→   157→        const match = result.find((i) => i.roomType.id === preset)

   159→   159→          setSelected(match)
   160→   160→          setStep(2)

   163→   163→            title: 'هذا النوع غير متاح للمواعيد المختارة',

   188→   188→        method: 'POST',

   190→   190→          checkIn: search.checkIn,
   191→   191→          checkOut: search.checkOut,
   192→   192→          adults: search.adults,
   193→   193→          children: search.children,
   194→   194→          roomsCount: search.roomsCount,
   195→   195→          roomTypeId: selected.roomType.id,
   196→   196→          guest: {
   197→   197→            fullName: guest.fullName.trim(),
   198→   198→            phone: guest.phone.trim(),
   199→   199→            ...(guest.whatsapp.trim() ? { whatsapp: guest.whatsapp.trim() } : {}),
   200→   200→            ...(guest.email.trim() ? { email: guest.email.trim() } : {}),
   201→   201→          },
   202→   202→          ...(guest.specialRequests.trim() ? { specialRequests: guest.specialRequests.trim() } : {}),
   203→   203→          paymentMethod,
   204→   204→          idempotencyKey: key,
   205→   205→        },

   208→   208→      setStep(5)
   209→   209→      toast({ title: 'تم تأكيد حجزك بنجاح', description: res.reservation.bookingReference })
   210→   210→    } catch (e) {
   211→   211→      const msg = e instanceof ApiError ? e.message : 'حدث خطأ غير متوقع أثناء الحجز'
   212→   212→      if (e instanceof ApiError && e.status === 409) {
   213→   213→        // فشل التوفر → عودة للخطوة 1 + تحديث النتائج
   214→   214→        setStep(1)
   215→   215→        toast({ title: 'تنبيه', description: msg })

   217→   217→        void runSearch(search)

   219→   219→        setStep(3)
   220→   220→        toast({ title: 'تعذر إتمام الحجز', description: msg, variant: 'destructive' })
   221→   221→      }
   222→   222→    } finally {
   223→   223→      submittingRef.current = false
   224→   224→    }
   225→   225→  }, [selected, hotel, search, guest, paymentMethod, runSearch, toast])
   226→   226→
   227→   227→  useEffect(() => {
   228→   228→    if (step === 4 && !submittingRef.current) {
   229→   229→      void submitBooking()
   230→   230→    }
   231→   231→  }, [step])
   232→   232→
   233→   233→  // ── إجراءات ──
   234→   234→  const submitSearchEdit = () => {
   235→   235→    if (search.checkOut <= search.checkIn) {
   236→   236→      setSearchError('تاريخ المغادرة يجب أن يكون بعد تاريخ الوصول')
   237→   237→      return
   238→   238→    }
   239→   239→    void runSearch(search)
   240→   240→    setEditOpen(false)
   241→   241→  }
   242→   242→
   243→   243→  const validateGuest = () => {
   244→   244→    const errs: Record<string, string> = {}
   245→   245→    if (guest.fullName.trim().length < 3) errs.fullName = 'يرجى إدخال الاسم الكامل (3 أحرف على الأقل)'
   246→   246→    if (digits(guest.phone).length < 9) errs.phone = 'يرجى إدخال رقم هاتف صحيح (9 أرقام على الأقل)'
   247→   247→    if (guest.whatsapp.trim() && digits(guest.whatsapp).length < 9)
   248→   248→      errs.whatsapp = 'رقم الواتساب غير صحيح (9 أرقام على الأقل)'
   249→   249→    if (guest.email.trim() && !/^\S+@\S+\.\S+$/.test(guest.email.trim()))
   250→   250→      errs.email = 'يرجى إدخال بريد إلكتروني صحيح أو تركه فارغًا'
   251→   251→    if (!policyAccepted) errs.policy = 'يرجى الإقرار بسياسة الإلغاء للمتابعة'
   252→   252→    setErrors(errs)
   253→   253→    return Object.keys(errs).length === 0
   254→   254→  }
   255→   255→
   256→   256→  const copyReference = async (ref: string) => {
   257→   257→    try {
   258→   258→      await navigator.clipboard.writeText(ref)
   259→   259→      toast({ title: 'تم النسخ', description: `تم نسخ رقم الحجز ${ref}` })
   260→   260→    } catch {
   261→   261→      toast({ title: 'تعذر النسخ', description: ref, variant: 'destructive' })
   262→   262→    }
   263→   263→  }
   264→   264→
   265→   265→  const closeDialog = () => onOpenChange(false)
   266→   266→
   267→   267→  return (
   268→   268→    <Dialog open={open} onOpenChange={onOpenChange}>
   269→   269→      <DialogContent className="flex max-h-[90vh] flex-col overflow-hidden gap-0 p-0 sm:max-w-3xl [&>button]:z-10">
   270→   270→        <DialogHeader className="border-b p-4 pb-3 sm:p-5 sm:pb-4">
   271→   271→          <DialogTitle className="text-lg font-extrabold sm:text-xl">
   272→   272→            {step === 5 ? 'تم الحجز' : 'احجز إقامتك'}
   273→   273→          </DialogTitle>
   274→   274→          <StepIndicator step={step} />
   275→   275→        </DialogHeader>
   276→   276→
   277→   277→        <div className="flex-1 overflow-y-auto p-4 sm:p-5">
   278→   278→          <AnimatePresence mode="wait">
   279→   279→            <motion.div
   280→   280→              key={step}
   281→   281→              initial={{ opacity: 0, x: -16 }}
   282→   282→              animate={{ opacity: 1, x: 0 }}
   283→   283→              exit={{ opacity: 0, x: 16 }}
   284→   284→              transition={{ duration: 0.22, ease: 'easeOut' }}
   285→   285→            >
   286→   286→              {step === 1 ? (
   287→   287→                <StepResults
   288→   288→                  search={search}
   289→   289→                  onSearchChange={setSearch}
   290→   290→                  items={items}
   291→   291→                  searching={searching}
   292→   292→                  error={searchError}
   293→   293→                  editOpen={editOpen}
   294→   294→                  onEditToggle={() => setEditOpen((o) => !o)}
   295→   295→                  onSearchSubmit={submitSearchEdit}
   296→   296→                  onRetry={() => void runSearch(search)}
   297→   297→                  onSelect={(item) => {
   298→   298→                    setSelected(item)
   299→   299→                    setStep(2)
   300→   300→                  }}
   301→   301→                />
   302→   302→              ) : null}
   303→   303→
   304→   304→              {step === 2 && selected ? (
   305→   305→                <StepGuest
   306→   306→                  guest={guest}
   307→   307→                  onChange={(g) => setGuest(g)}
   308→   308→                  errors={errors}
   309→   309→                  hotel={hotel}
   310→   310→                  policyAccepted={policyAccepted}
   311→   311→                  onPolicyChange={(c) => setPolicyAccepted(c)}
   312→   312→                  onBack={() => setStep(1)}
   313→   313→                  onSubmit={() => {
   314→   314→                    if (validateGuest()) setStep(3)
   315→   315→                  }}
   316→   316→                />
   317→   317→              ) : null}
   318→   318→
   319→   319→              {step === 3 && selected ? (
   320→   320→                <StepReview
   321→   321→                  hotel={hotel}
   322→   322→                  selected={selected}
   323→   323→                  search={search}
   324→   324→                  guest={guest}
   325→   325→                  paymentMethod={paymentMethod}
   326→   326→                  onPaymentChange={setPaymentMethod}
   327→   327→                  onBack={() => setStep(2)}
   328→   328→                  onConfirm={() => setStep(4)}
   329→   329→                />
   330→   330→              ) : null}
   331→   331→
   332→   332→              {step === 4 ? <StepProcessing /> : null}
   333→   333→
   334→   334→              {step === 5 && confirmed ? (
   335→   335→                <StepConfirmation
   336→   336→                  hotel={hotel}
   337→   337→                  reservation={confirmed}
   338→   338→                  snapshot={selected ? snapshotFromQuote(selected, hotel) : null}
   339→   339→                  paymentMethod={paymentMethod}
   340→   340→                  onCopy={copyReference}
   341→   341→                  onPrint={() =>
   342→   342→                    onPrint({
   343→   343→                      reservation: confirmed,
   344→   344→                      snapshot: selected ? snapshotFromQuote(selected, hotel) : null,
   345→   345→                    })
   346→   346→                  }
   347→   347→                  onManage={() => {
   348→   348→                    onManageBooking(confirmed.bookingReference)
   349→   349→                  }}
   350→   350→                  onClose={closeDialog}
   351→   351→                />
   352→   352→              ) : null}
   353→   353→            </motion.div>
   354→   354→          </AnimatePresence>
   355→   355→        </div>
   356→   356→      </DialogContent>
   357→   357→    </Dialog>
   358→   358→  )
   359→   359→}

   376→   376→function StepIndicator({ step }: { step: number }) {
   377→   377→  return (
   378→   378→    <ol className="mt-3 flex items-center gap-1.5" aria-label="خطوات الحجز">
   379→   379→      {STEP_LABELS.map((label, i) => {
   380→   380→        const n = i + 1
   381→   381→        const done = n < step
   382→   382→        const active = n === step
   383→   383→        return (
   384→   384→          <li key={label} className="flex flex-1 items-center gap-1.5">
   385→   385→            <span
   386→   386→              className={`flex size-6 shrink-0 items-center justify-center rounded-full text-[11px] font-black transition-colors ${
   387→   387→                done
   388→   388→                  ? 'bg-success text-white'
   389→   389→                  : active
   390→   390→                    ? 'bg-primary text-primary-foreground'
   391→   391→                    : 'bg-muted text-muted-foreground'
   392→   392→              }`}
   393→   393→              aria-current={active ? 'step' : undefined}
   394→   394→            >
   395→   395→              {done ? <CheckCircle2 className="size-4" /> : n}
   396→   396→            </span>
   397→   397→            <span
   398→   398→              className={`hidden text-[11px] font-bold sm:inline ${
   399→   399→                active ? 'text-foreground' : 'text-muted-foreground'
   400→   400→              }`}
   401→   401→            >
   402→   402→              {label}
   403→   403→            </span>
   404→   404→            {n < STEP_LABELS.length ? (
   405→   405→              <span className={`h-px flex-1 ${done ? 'bg-success/60' : 'bg-border'}`} />
   406→   406→            ) : null}
   407→   407→          </li>
   408→   408→        )
   409→   409→      })}
   410→   410→    </ol>
   411→   411→  )
   412→   412→}

   416→   416→function StepResults({
   417→   417→  search,
   418→   418→  onSearchChange,
   419→   419→  items,
   420→   420→  searching,
   421→   421→  error,
   422→   422→  editOpen,
   423→   423→  onEditToggle,
   424→   424→  onSearchSubmit,
   425→   425→  onRetry,
   426→   426→  onSelect,
   427→   427→}: {
   428→   428→  search: SearchParams
   429→   429→  onSearchChange: (s: SearchParams) => void
   430→   430→  items: AvailabilityItem[]
   431→   431→  searching: boolean
   432→   432→  error: string | null
   433→   433→  editOpen: boolean
   434→   434→  onEditToggle: () => void
   435→   435→  onSearchSubmit: () => void
   436→   436→  onRetry: () => void
   437→   437→  onSelect: (item: AvailabilityItem) => void
   438→   438→}) {
   439→   439→  const set = (patch: Partial<SearchParams>) => {
   440→   440→    const next = { ...search, ...patch }
   441→   441→    if (next.checkOut <= next.checkIn) next.checkOut = addDaysInput(next.checkIn, 1)
   442→   442→    onSearchChange(next)
   443→   443→  }
   444→   444→
   445→   445→  return (
   446→   446→    <div className="space-y-4">
   447→   447→      {/* ملخص البحث */}
   448→   448→      <div className="flex flex-wrap items-center justify-between gap-2 rounded-lg border bg-muted/50 px-3 py-2">
   449→   449→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-sm font-semibold text-foreground">
