    57→  phone: string
    58→  whatsapp: string

    63→const EMPTY_GUEST: GuestForm = { fullName: '', phone: '', whatsapp: '', email: '', specialRequests: '' }

    65→interface BookingDialogProps {
    66→  open: boolean
    67→  onOpenChange: (open: boolean) => void
    68→  hotel: HotelPublic | null
    69→  initialSearch: SearchParams
    70→  presetRoomTypeId?: string

    75→export function BookingDialog({
    76→  open,
    77→  onOpenChange,
    78→  hotel,
    79→  initialSearch,
    80→  presetRoomTypeId,
    81→  onManageBooking,
    82→  onPrint,
    83→}: BookingDialogProps) {
    84→  const { toast } = useToast()
    85→
    86→  const [step, setStep] = useState(1)
    87→  const [search, setSearch] = useState<SearchParams>(initialSearch)
    88→  const [editOpen, setEditOpen] = useState(false)
    89→  const [items, setItems] = useState<AvailabilityItem[]>([])
    90→  const [searching, setSearching] = useState(false)
    91→  const [searchError, setSearchError] = useState<string | null>(null)
    92→  const [selected, setSelected] = useState<AvailabilityItem | null>(null)
    93→  const [guest, setGuest] = useState<GuestForm>(EMPTY_GUEST)
    94→  const [policyAccepted, setPolicyAccepted] = useState(false)
    95→  const [errors, setErrors] = useState<Record<string, string>>({})
    96→  const [paymentMethod, setPaymentMethod] = useState<'PAY_AT_HOTEL' | 'CARD'>('PAY_AT_HOTEL')
    97→  const [confirmed, setConfirmed] = useState<ReservationPublic | null>(null)
    98→
    99→  // آخر قيم props (تُقرأ عند الفتح دون إعادة تشغيل التأثيرات)
   100→  const initialSearchRef = useRef(initialSearch)
   101→  initialSearchRef.current = initialSearch
   102→  const presetRef = useRef(presetRoomTypeId)
   103→  presetRef.current = presetRoomTypeId
   104→
   105→  // مفتاح idempotency — يولد مرة لكل تدفق حجز ويثبت
   106→  const idemKeyRef = useRef('')
   107→  const submittingRef = useRef(false)
   108→
   109→  // ── البحث ──
   110→  const runSearch = useCallback(
   111→    async (s: SearchParams): Promise<AvailabilityItem[]> => {
   112→      setSearching(true)
   113→      setSearchError(null)
   114→      try {
   115→        const res = await api<{ items: AvailabilityItem[] }>('/api/public/availability', {
   116→          method: 'POST',
   117→          body: s,
   118→        })
   119→        setItems(res.items)
   120→        return res.items
   121→      } catch (e) {
   122→        setItems([])
   123→        const msg = e instanceof ApiError ? e.message : 'تعذر البحث عن الغرف — يرجى المحاولة مرة أخرى'
   124→        setSearchError(msg)
   125→        return []
   126→      } finally {
   127→        setSearching(false)
   128→      }
   129→    },
   130→    []
   131→  )
   132→
   133→  // ── إعادة التعيين + البحث التلقائي عند الفتح ──
   134→  useEffect(() => {
   135→    if (!open) return
   136→    const s = initialSearchRef.current
   137→    const preset = presetRef.current
   138→    setStep(1)
   139→    setSearch(s)
   140→    setEditOpen(false)
   141→    setItems([])
   142→    setSearchError(null)
   143→    setSelected(null)
   144→    setGuest(EMPTY_GUEST)
   145→    setPolicyAccepted(false)
   146→    setErrors({})
   147→    setPaymentMethod('PAY_AT_HOTEL')
   148→    setConfirmed(null)
   149→    idemKeyRef.current = ''
   150→    submittingRef.current = false
   151→
   152→    let cancelled = false
   153→    void runSearch(s).then((result) => {
   154→      if (cancelled) return
   155→      // تحديد مسبق لنوع الغرفة (زر «احجز الآن» من بطاقة الغرفة)
   156→      if (preset) {
   157→        const match = result.find((i) => i.roomType.id === preset)
   158→        if (match) {
   159→          setSelected(match)
   160→          setStep(2)
   161→        } else if (result.length > 0) {
   162→          toast({
   163→            title: 'هذا النوع غير متاح للمواعيد المختارة',
   164→            description: 'اختر أحد الخيارات المتاحة أدناه أو عدّل المواعيد',
   165→          })
   166→        }
   167→      }
   168→    })
   169→    return () => {
   170→      cancelled = true
   171→    }
   172→  }, [open])
   173→
   174→  // ── الخطوة 4: إرسال الحجز تلقائيًا (مرة واحدة لكل دخول) ──
   175→  const submitBooking = useCallback(async () => {
   176→    if (!selected || !hotel) return
   177→    submittingRef.current = true
   178→    if (!idemKeyRef.current) {
   179→      idemKeyRef.current =
   180→        typeof crypto !== 'undefined' && 'randomUUID' in crypto
   181→          ? crypto.randomUUID()
   182→          : `idem-${Date.now()}-${Math.random().toString(36).slice(2)}`
   183→    }
   184→    const key = idemKeyRef.current
   185→
   186→    try {
   187→      const res = await api<{ reservation: ReservationPublic }>('/api/public/bookings', {
   188→        method: 'POST',
   189→        body: {
   190→          checkIn: search.checkIn,
   191→          checkOut: search.checkOut,
   192→          adults: search.adults,
   193→          children: search.children,
   194→          roomsCount: search.roomsCount,
   195→          roomTypeId: selected.roomType.id,
   196→          guest: {
   197→            fullName: guest.fullName.trim(),
   198→            phone: guest.phone.trim(),
   199→            ...(guest.whatsapp.trim() ? { whatsapp: guest.whatsapp.trim() } : {}),
   200→            ...(guest.email.trim() ? { email: guest.email.trim() } : {}),
   201→          },
   202→          ...(guest.specialRequests.trim() ? { specialRequests: guest.specialRequests.trim() } : {}),
   203→          paymentMethod,
   204→          idempotencyKey: key,
   205→        },
   206→      })
   207→      setConfirmed(res.reservation)
   208→      setStep(5)
   209→      toast({ title: 'تم تأكيد حجزك بنجاح', description: res.reservation.bookingReference })
   210→    } catch (e) {
   211→      const msg = e instanceof ApiError ? e.message : 'حدث خطأ غير متوقع أثناء الحجز'
   212→      if (e instanceof ApiError && e.status === 409) {
   213→        // فشل التوفر → عودة للخطوة 1 + تحديث النتائج
   214→        setStep(1)
   215→        toast({ title: 'تنبيه', description: msg })
   216→        setSelected(null)
   217→        void runSearch(search)
   218→      } else {
   219→        setStep(3)
   220→        toast({ title: 'تعذر إتمام الحجز', description: msg, variant: 'destructive' })
   221→      }
   222→    } finally {
   223→      submittingRef.current = false
   224→    }
   225→  }, [selected, hotel, search, guest, paymentMethod, runSearch, toast])
   226→
   227→  useEffect(() => {
   228→    if (step === 4 && !submittingRef.current) {
   229→      void submitBooking()
   230→    }
   231→  }, [step])
   232→
   233→  // ── إجراءات ──
   234→  const submitSearchEdit = () => {
   235→    if (search.checkOut <= search.checkIn) {
   236→      setSearchError('تاريخ المغادرة يجب أن يكون بعد تاريخ الوصول')
   237→      return
   238→    }
   239→    void runSearch(search)
   240→    setEditOpen(false)
   241→  }
   242→
   243→  const validateGuest = () => {
   244→    const errs: Record<string, string> = {}
   245→    if (guest.fullName.trim().length < 3) errs.fullName = 'يرجى إدخال الاسم الكامل (3 أحرف على الأقل)'
   246→    if (digits(guest.phone).length < 9) errs.phone = 'يرجى إدخال رقم هاتف صحيح (9 أرقام على الأقل)'
   247→    if (guest.whatsapp.trim() && digits(guest.whatsapp).length < 9)
   248→      errs.whatsapp = 'رقم الواتساب غير صحيح (9 أرقام على الأقل)'
   249→    if (guest.email.trim() && !/^\S+@\S+\.\S+$/.test(guest.email.trim()))
   250→      errs.email = 'يرجى إدخال بريد إلكتروني صحيح أو تركه فارغًا'
   251→    if (!policyAccepted) errs.policy = 'يرجى الإقرار بسياسة الإلغاء للمتابعة'
   252→    setErrors(errs)
   253→    return Object.keys(errs).length === 0
   254→  }
   255→
   256→  const copyReference = async (ref: string) => {
   257→    try {
   258→      await navigator.clipboard.writeText(ref)
   259→      toast({ title: 'تم النسخ', description: `تم نسخ رقم الحجز ${ref}` })
   260→    } catch {
   261→      toast({ title: 'تعذر النسخ', description: ref, variant: 'destructive' })
   262→    }
   263→  }
   264→
   265→  const closeDialog = () => onOpenChange(false)
   266→
   267→  return (
   268→    <Dialog open={open} onOpenChange={onOpenChange}>
   269→      <DialogContent className="flex max-h-[90vh] flex-col overflow-hidden gap-0 p-0 sm:max-w-3xl [&>button]:z-10">
   270→        <DialogHeader className="border-b p-4 pb-3 sm:p-5 sm:pb-4">
   271→          <DialogTitle className="text-lg font-extrabold sm:text-xl">
   272→            {step === 5 ? 'تم الحجز' : 'احجز إقامتك'}
   273→          </DialogTitle>
   274→          <StepIndicator step={step} />
   275→        </DialogHeader>
   276→
   277→        <div className="flex-1 overflow-y-auto p-4 sm:p-5">
   278→          <AnimatePresence mode="wait">
   279→            <motion.div
   280→              key={step}
   281→              initial={{ opacity: 0, x: -16 }}
   282→              animate={{ opacity: 1, x: 0 }}
   283→              exit={{ opacity: 0, x: 16 }}
   284→              transition={{ duration: 0.22, ease: 'easeOut' }}
   285→            >
   286→              {step === 1 ? (
   287→                <StepResults
   288→                  search={search}
   289→                  onSearchChange={setSearch}
   290→                  items={items}
   291→                  searching={searching}
   292→                  error={searchError}
   293→                  editOpen={editOpen}
   294→                  onEditToggle={() => setEditOpen((o) => !o)}
   295→                  onSearchSubmit={submitSearchEdit}
   296→                  onRetry={() => void runSearch(search)}
   297→                  onSelect={(item) => {
   298→                    setSelected(item)
   299→                    setStep(2)
   300→                  }}
   301→                />
   302→              ) : null}
   303→
   304→              {step === 2 && selected ? (
   305→                <StepGuest
   306→                  guest={guest}
   307→                  onChange={(g) => setGuest(g)}
   308→                  errors={errors}
   309→                  hotel={hotel}
   310→                  policyAccepted={policyAccepted}
   311→                  onPolicyChange={(c) => setPolicyAccepted(c)}
   312→                  onBack={() => setStep(1)}
   313→                  onSubmit={() => {
   314→                    if (validateGuest()) setStep(3)
   315→                  }}
   316→                />
   317→              ) : null}
   318→
   319→              {step === 3 && selected ? (
   320→                <StepReview
   321→                  hotel={hotel}
   322→                  selected={selected}
   323→                  search={search}
   324→                  guest={guest}
   325→                  paymentMethod={paymentMethod}
   326→                  onPaymentChange={setPaymentMethod}
   327→                  onBack={() => setStep(2)}
   328→                  onConfirm={() => setStep(4)}
   329→                />
   330→              ) : null}
   331→
   332→              {step === 4 ? <StepProcessing /> : null}
   333→
   334→              {step === 5 && confirmed ? (
   335→                <StepConfirmation
   336→                  hotel={hotel}
   337→                  reservation={confirmed}
   338→                  snapshot={selected ? snapshotFromQuote(selected, hotel) : null}
   339→                  paymentMethod={paymentMethod}
   340→                  onCopy={copyReference}
   341→                  onPrint={() =>
   342→                    onPrint({
   343→                      reservation: confirmed,
   344→                      snapshot: selected ? snapshotFromQuote(selected, hotel) : null,
   345→                    })
   346→                  }
   347→                  onManage={() => {
   348→                    onManageBooking(confirmed.bookingReference)
   349→                  }}
   350→                  onClose={closeDialog}
   351→                />
   352→              ) : null}
   353→            </motion.div>
   354→          </AnimatePresence>
   355→        </div>
   356→      </DialogContent>
   357→    </Dialog>
   358→  )
   359→}

   361→/** بناء لقطة عرض من عرض السعر (الملخص فقط — المجاميع من استجابة الخادم) */
   362→function snapshotFromQuote(item: AvailabilityItem, hotel: HotelPublic | null): SnapshotBreakdown {
   363→  return {
   364→    nightly: item.quote.nightly,
   365→    subtotalCents: item.quote.subtotalCents,
   366→    taxCents: item.quote.taxCents,
   367→    grandTotalCents: item.quote.grandTotalCents,
   368→    cancellationPolicy: hotel?.cancellationPolicy,
   369→    checkInTime: hotel?.checkInTime,
   370→    checkOutTime: hotel?.checkOutTime,
   371→  }
   372→}
   373→
   374→// ─────────────────────────── مؤشر الخطوات ───────────────────────────
   375→
   376→function StepIndicator({ step }: { step: number }) {
   377→  return (
   378→    <ol className="mt-3 flex items-center gap-1.5" aria-label="خطوات الحجز">
   379→      {STEP_LABELS.map((label, i) => {
   380→        const n = i + 1
   381→        const done = n < step
   382→        const active = n === step
   383→        return (
   384→          <li key={label} className="flex flex-1 items-center gap-1.5">
   385→            <span
   386→              className={`flex size-6 shrink-0 items-center justify-center rounded-full text-[11px] font-black transition-colors ${
   387→                done
   388→                  ? 'bg-success text-white'
   389→                  : active
   390→                    ? 'bg-primary text-primary-foreground'
   391→                    : 'bg-muted text-muted-foreground'
   392→              }`}
   393→              aria-current={active ? 'step' : undefined}
   394→            >
   395→              {done ? <CheckCircle2 className="size-4" /> : n}
   396→            </span>
   397→            <span
   398→              className={`hidden text-[11px] font-bold sm:inline ${
   399→                active ? 'text-foreground' : 'text-muted-foreground'
   400→              }`}
   401→            >
   402→              {label}
   403→            </span>
   404→            {n < STEP_LABELS.length ? (
   405→              <span className={`h-px flex-1 ${done ? 'bg-success/60' : 'bg-border'}`} />
   406→            ) : null}
   407→          </li>
   408→        )
   409→      })}
   410→    </ol>
   411→  )
   412→}

   414→// ─────────────────────────── الخطوة 1: النتائج ───────────────────────────

   416→function StepResults({
   417→  search,
   418→  onSearchChange,
   419→  items,
   420→  searching,
   421→  error,
   422→  editOpen,
   423→  onEditToggle,
   424→  onSearchSubmit,
   425→  onRetry,
   426→  onSelect,
   427→}: {
   428→  search: SearchParams
   429→  onSearchChange: (s: SearchParams) => void
   430→  items: AvailabilityItem[]
   431→  searching: boolean
   432→  error: string | null
   433→  editOpen: boolean
   434→  onEditToggle: () => void
   435→  onSearchSubmit: () => void
   436→  onRetry: () => void
   437→  onSelect: (item: AvailabilityItem) => void
   438→}) {
   439→  const set = (patch: Partial<SearchParams>) => {
   440→    const next = { ...search, ...patch }
   441→    if (next.checkOut <= next.checkIn) next.checkOut = addDaysInput(next.checkIn, 1)
   442→    onSearchChange(next)
   443→  }
   444→
   445→  return (
   446→    <div className="space-y-4">
   447→      {/* ملخص البحث */}
   448→      <div className="flex flex-wrap items-center justify-between gap-2 rounded-lg border bg-muted/50 px-3 py-2">
   449→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-sm font-semibold text-foreground">
   450→          <span className="inline-flex items-center gap-1.5">
   451→            <CalendarDays className="size-4 text-primary dark:text-gold" />
   452→            {formatDateAr(search.checkIn)} ← {formatDateAr(search.checkOut)}
   453→          </span>
   454→          <span className="inline-flex items-center gap-1.5">
   455→            <Users className="size-4 text-primary dark:text-gold" />
   456→            {guestsText(search.adults, search.children)}
   457→          </span>
   458→          <span className="inline-flex items-center gap-1.5">
   459→            <BedDouble className="size-4 text-primary dark:text-gold" />
   460→            {roomsText(search.roomsCount)}
   461→          </span>
   462→        </div>
   463→        <Button variant="outline" size="sm" onClick={onEditToggle}>
   464→          <Pencil className="size-3.5" />
   465→          {editOpen ? 'إخفاء' : 'تعديل'}
   466→        </Button>
   467→      </div>
   468→
   469→      {/* نموذج التعديل */}
   470→      {editOpen ? (
   471→        <div className="grid grid-cols-2 gap-3 rounded-lg border p-3 md:grid-cols-3 xl:grid-cols-6">
   472→          <div className="space-y-1.5">
   473→            <Label className="text-xs text-muted-foreground">الوصول</Label>
   474→            <Input type="date" dir="ltr" value={search.checkIn} min={todayInputValue()} onChange={(e) => set({ checkIn: e.target.value })} />
   475→          </div>
   476→          <div className="space-y-1.5">
   477→            <Label className="text-xs text-muted-foreground">المغادرة</Label>
   478→            <Input
   479→              type="date"
   480→              dir="ltr"
   481→              value={search.checkOut}
   482→              min={addDaysInput(search.checkIn, 1)}
   483→              onChange={(e) => set({ checkOut: e.target.value })}
   484→            />
   485→          </div>
   486→          <div className="space-y-1.5">
   487→            <Label className="text-xs text-muted-foreground">البالغون</Label>
   488→            <Select value={String(search.adults)} onValueChange={(v) => set({ adults: Number(v) })}>
   489→              <SelectTrigger dir="rtl">
   490→                <SelectValue />
   491→              </SelectTrigger>
   492→              <SelectContent>
   493→                {[1, 2, 3, 4, 5, 6].map((n) => (
   494→                  <SelectItem key={n} value={String(n)}>
   495→                    {n === 1 ? 'بالغ واحد' : n === 2 ? 'بالغان' : `${n} بالغين`}
   496→                  </SelectItem>
   497→                ))}
   498→              </SelectContent>
   499→            </Select>
   500→          </div>
   501→          <div className="space-y-1.5">
   502→            <Label className="text-xs text-muted-foreground">الأطفال</Label>
   503→            <Select value={String(search.children)} onValueChange={(v) => set({ children: Number(v) })}>
   504→              <SelectTrigger dir="rtl">
   505→                <SelectValue />
   506→              </SelectTrigger>
   507→              <SelectContent>
   508→                {[0, 1, 2, 3, 4].map((n) => (
   509→                  <SelectItem key={n} value={String(n)}>
   510→                    {n === 0 ? 'بدون' : n === 1 ? 'طفل واحد' : n === 2 ? 'طفلان' : `${n} أطفال`}
   511→                  </SelectItem>
   512→                ))}
   513→              </SelectContent>
   514→            </Select>
   515→          </div>
   516→          <div className="space-y-1.5">
   517→            <Label className="text-xs text-muted-foreground">الغرف</Label>
   518→            <Select value={String(search.roomsCount)} onValueChange={(v) => set({ roomsCount: Number(v) })}>
   519→              <SelectTrigger dir="rtl">
   520→                <SelectValue />
   521→              </SelectTrigger>
   522→              <SelectContent>
   523→                {[1, 2, 3].map((n) => (
   524→                  <SelectItem key={n} value={String(n)}>
   525→                    {n === 1 ? 'غرفة واحدة' : n === 2 ? 'غرفتان' : `${n} غرف`}
   526→                  </SelectItem>
   527→                ))}
   528→              </SelectContent>
